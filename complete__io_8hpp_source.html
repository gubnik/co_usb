67 template <detail::AnyBufferSequence BuffersTy>
68 auto complete_submit (BuffersTy const &buffers) -> boost::capy::io_task<size_t>
69 {
70 auto exec = co_await boost::capy::this_coro::executor;
71 auto stop = co_await boost::capy::this_coro::stop_token;
72
73 const size_t transfers_used =
74 std::min(boost::capy::buffer_length(buffers), this->m_view.size());
75 std::stop_callback sb{stop, [&] ()
76 {
77 const auto end = this->m_view.begin() + transfers_used;
78 for (auto iter = this->m_view.begin(); iter != end; iter++)
79 {
80 libusb_transfer *tfer = detail::transfer_of(*iter);
81 cancel_transfer(tfer);
82 }
83 }};
84 using awaitable_t = detail::complete_sequence_awaitable<TSeq, BuffersTy>;
85 typename awaitable_t::await_state_t await_state(m_memres);
86 co_return co_await awaitable_t(&await_state, &this->m_view, buffers, transfers_used,
87 m_single_transfer_limit);
88 }
89
90 private:
91 std::pmr::memory_resource *m_memres;
92 size_t m_single_transfer_limit;
93};
