73 template <detail::AnyBufferSequence BuffersTy>
74 auto submit (BuffersTy const &buffers) -> boost::capy::io_task<size_t>
75 {
76 auto exec = co_await boost::capy::this_coro::executor;
77 auto stop = co_await boost::capy::this_coro::stop_token;
78
79 const size_t transfers_used = std::min(boost::capy::buffer_length(buffers), m_view.size());
80 std::stop_callback sb{stop, [&] ()
81 {
82 const auto end = m_view.begin() + transfers_used;
83 for (auto iter = m_view.begin(); iter != end; iter++)
84 {
85 libusb_transfer *tfer = detail::transfer_of(*iter);
86 cancel_transfer(tfer);
87 }
88 }};
89 using awaitable_t = detail::sequence_awaitable<TSeq, BuffersTy>;
90 typename awaitable_t::await_state_t await_state;
91 co_return co_await awaitable_t(&await_state, &m_view, buffers, transfers_used);
92 }
93
94 protected:
95 ev::event_handler_ref m_ev_ref;
96 sequence_view<TSeq> m_view;
97};
